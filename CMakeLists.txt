cmake_minimum_required(VERSION 3.22)
project(OpenDeHISS VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT JUCE_SOURCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../juce-src/CMakeLists.txt")
  set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../juce-src")
endif()
if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/91ad83ae34a81e0833b1a2b0866f54846370ae53.zip
    URL_HASH SHA256=FA9486B18C3624BC74A8146BFAD4AA100101F210849F1F65B44F4011096B47AE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(OpenDeHISS COMPANY_NAME "AudioCheck" PLUGIN_MANUFACTURER_CODE AuCh
  PLUGIN_CODE Odh1 FORMATS VST3 PRODUCT_NAME "Open DeHISS"
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
  NEEDS_WEB_BROWSER FALSE NEEDS_CURL FALSE)
target_sources(OpenDeHISS PRIVATE Processor.cpp Editor.cpp)
target_compile_definitions(OpenDeHISS PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(OpenDeHISS PRIVATE juce::juce_audio_utils juce::juce_dsp)
juce_add_console_app(OpenDeHISSTests PRODUCT_NAME "Open DeHISS Tests")
target_sources(OpenDeHISSTests PRIVATE Tests.cpp)
target_link_libraries(OpenDeHISSTests PRIVATE OpenDeHISS juce::juce_audio_utils juce::juce_dsp)
enable_testing()
add_test(NAME dsp_and_editor COMMAND OpenDeHISSTests)
option(OPEN_DEHISS_BUILD_HOST_CHECK "Build isolated VST3 host verification" OFF)
if(OPEN_DEHISS_BUILD_HOST_CHECK)
  juce_add_console_app(OpenDeHISSHost PRODUCT_NAME "Open DeHISS Host Check")
  target_sources(OpenDeHISSHost PRIVATE tools/HostCheck.cpp)
  target_compile_definitions(OpenDeHISSHost PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(OpenDeHISSHost PRIVATE juce::juce_audio_processors_headless)
  add_test(NAME vst3_host COMMAND OpenDeHISSHost "${CMAKE_BINARY_DIR}/OpenDeHISS_artefacts/$<CONFIG>/VST3/Open DeHISS.vst3")
endif()
option(OPEN_DEHISS_BUILD_REFERENCE_HOST "Build Windows reference VST3 benchmark host" OFF)
if(OPEN_DEHISS_BUILD_REFERENCE_HOST AND WIN32)
  juce_add_console_app(OpenDeHISSReferenceHost PRODUCT_NAME "Open DeHISS Reference Host")
  target_sources(OpenDeHISSReferenceHost PRIVATE tools/ReferenceHost.cpp)
  target_compile_definitions(OpenDeHISSReferenceHost PRIVATE REFERENCE_WITH_EDITOR=0 JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(OpenDeHISSReferenceHost PRIVATE juce::juce_audio_processors_headless juce::juce_audio_formats)
  add_test(NAME reference_host_load COMMAND OpenDeHISSReferenceHost "${CMAKE_BINARY_DIR}/OpenDeHISS_artefacts/$<CONFIG>/VST3/Open DeHISS.vst3")
endif()
